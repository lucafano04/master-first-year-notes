\chapter{Performance Metrics [14--16/09/2026]} 
This chapter explores the fundamental performance metrics, power scaling constraints, cost evaluation models, and quantitative design principles utilized in computer architecture.

\section{Power Consumption}
In modern \gls{cmos} technology, power consumption is driven by two primary components:
\begin{description}
    \item[Dynamic (Active)] Currents related to:
    \begin{itemize}
        \item The charging and discharging of logic gates during switching states.
        \item Transient short-circuit currents during signal transitions.
    \end{itemize}
    \item[Static] Leakage (sub-threshold) currents flowing continuously through transistors.
\end{description}
\noindent The prevailing historical trend shows that with each process generation, dynamic power scales by a factor of roughly $0.9$, while power density increases by a factor of $1.3$ to $1.8$, and static power consumption grows at an exponential rate.

\subsection{Dynamic Power Consumption}
\begin{definition}{Dynamic Power Consumption}{dyn-power}
    Dynamic power consumption is primarily caused by the charging and discharging of the capacitive load of every logic gate inside a digital \gls{cmos} integrated circuit:
    \begin{align*}
        \text{Energy/transition} &= C_L \cdot V_{DD}^2 \cdot P_{0\rightarrow1} \\
        \text{Power} &= \text{Energy/transition} \cdot f = C_{L} \cdot V_{DD}^2 \cdot P_{0\rightarrow1} \cdot f
    \end{align*} 
    Where:
    \begin{description}
        \item[\(C_L\) (Capacitance):] A function of wire length and transistor size.
        \item[\(V_{DD}\) (Supply Voltage):] Has steadily decreased across successive fabrication generations.
        \item[\(P_{0\rightarrow1}\) (Switching Probability):] The statistical probability of a logic transition.
        \item[\(f\) (Clock Frequency):] Higher frequencies increase the rate of gate state changes, raising power dissipation.
    \end{description}
\end{definition}

\begin{definition}{Short-Circuit Current}{scc}
    The finite slope of input signals causes a direct current path between \(V_{DD}\) and \gls{gnd} for a brief period during switching when both the \gls{nmos} and \gls{pmos} transistors are simultaneously conducting:
    \begin{align*}
        E_{sc} &= t_{sc} \cdot V_{DD} \cdot I_{\text{peak}} \cdot P_{0\rightarrow 1}\\
        P_{sc} &= t_{sc} \cdot V_{DD} \cdot I_{\text{peak}} \cdot f_{0\rightarrow 1}
    \end{align*}
    Where \(t_{sc}\) represents the rise/fall duration of the input signal, and \(I_{\text{peak}}\) is determined by the saturation current of the transistors and the input-to-output slope ratio.
\end{definition}

\subsection{Static Power Consumption}
\begin{definition}{Static Power Consumption}{stat-power}
    Sub-threshold current grows exponentially with an increase in temperature and a decrease in threshold voltage. As supply voltage scaling continues, sub-threshold conduction becomes the primary source of power dissipation:
    \[
        I_{\text{DSub}} = k \cdot e^{\frac{-q \cdot V_{T}}{a \cdot k_a \cdot T}}
    \]
    where \(k, a, k_a\) are technology-dependent constants, \(V_T\) is the threshold voltage, and \(T\) is the absolute temperature.
\end{definition}

\subsection{Total Power Consumption}
\begin{definition}{Total Power Consumption}{tot-power}
    Combining dynamic, short-circuit, and leakage components, the total power consumption of a system is expressed as:
    \[
        P_{\text{total}} = \left(C_{L} \cdot V^2_{DD} \cdot P_{0\rightarrow1} \cdot f\right) + \left(t_{sc} \cdot V_{DD} \cdot P_{0\rightarrow1} \cdot I_{\text{peak}} \cdot f\right) + \left(V_{DD} \cdot I_{\text{leakage}}\right)
    \]
    At the architectural level, designers manage these levers: \(f_{\text{clock}}\) (trading off performance), \(C_L\) (reducing gate counts), \(V_{DD}\) (bounded by transistor precision limits), and \(P_{0\rightarrow1}\) (optimized via microarchitectural design).
\end{definition}

\section{Cost Estimation}
Device cost tends to decrease over time even without improvements in fabrication technology, driven by the \textbf{learning curve effect}. By increasing production volumes, manufacturing costs drop significantly ($\sim 10\%$ cost reduction per doubling of volume), allowing non-recurring engineering (\gls{nre}) and development costs to be amortized over a massive machine base.

\begin{definition}{Integrated Circuit Cost}{ic-cost}
    The cost of a manufactured \gls{ic} is governed by:
    \[
        C = \frac{C_{\text{die}} + C_{\text{testing}} + C_{\text{packaging}}}{Y}
    \]
    where \(C_{\text{die}}\) is the manufacturing cost of the die, \(C_{\text{testing}}\) and \(C_{\text{packaging}}\) are fixed post-processing costs dependent on complexity, and \(Y\) is the final manufacturing yield.
\end{definition}

\begin{definition}{Die Cost and Wafer Capacity}{die-cost}
    Single-die cost depends directly on wafer yield and total available dies:
    \[
        C_{\text{die}} = \frac{C_{\text{wafer}}}{N_{\text{die}} \cdot Y_{\text{die}}}
    \]
    The total number of dies per wafer (\(N_{\text{die}}\)) is calculated accounting for wafer circular geometry and peripheral edge loss:
    \[
        N_{\text{die}} = \frac{\pi \cdot D^2}{4 \cdot A_{\text{die}}} - \frac{\pi \cdot D}{\sqrt{2 \cdot A_{\text{die}}}}
    \]
    This highlights the strong, non-linear cost penalty associated with larger die areas (\(A_{\text{die}}\)).
\end{definition}

\begin{definition}{Die Yield}{die-yield}
    Pre-packaging die yield is modeled statistically as:
    \[
        Y_{\text{die}} = Y_{\text{wafer}} \cdot \left(1 + \frac{N_{\text{defects}} \cdot A_{\text{die}}}{\alpha}\right)^{-\alpha}
    \]
    where \(N_{\text{defects}}\) is defect density per unit area, and \(\alpha\) ($\sim 4.0$) is an empirical parameter reflecting process complexity.
\end{definition}

\section{Performance Evaluation}
Historically, microprocessor performance scaling was driven by \textbf{Moore's Law} (\cref{cpt:moores-law}). Performance today scales faster than transistor density alone due to architectural parallelism.

\begin{concept}{Moore's Law}{moores-law}
    An empirical observation stating that the number of transistors in a dense \gls{ic} historically doubles approximately every 15 to 18 months.
\end{concept}

\begin{definition}{Ideal Properties of a Performance Metric}{perf_metric_properties}
    A robust performance metric must satisfy several key criteria: \textbf{Linearity} (proportional scale matching), \textbf{Reliability} (consistent relative ranking of systems \(X\) and \(Y\)), \textbf{Repeatability} (compatible variance across test runs), \textbf{Ease of Use}, \textbf{Consistency} (standardized definitions), and \textbf{Independence} (unambiguous mathematical formulation).
\end{definition}

\subsection{The Clock Rate}
While clock rate is repeatable, easy to measure, consistent, and independent, it is a **misleading metric** (pushed primarily for marketing). It is neither linear nor reliable because it ignores instructions per cycle, computational complexity per cycle, and memory/I/O bottlenecks.

\subsection{Instruction and Operation Metrics}
\begin{definition}{MIPS (Million Instructions Per Second)}{mips-def}
    Defined as the execution instruction throughput:
    \[
        \text{\gls{mips}} = \frac{N}{T \cdot 10^6}
    \]
    where \(N\) is total instruction count and \(T\) is execution time. While easy to measure, it is non-linear and inconsistent across different \glspl{isa}.
\end{definition}

\begin{definition}{MOPS and MFLOPS}{mops-mflops-def}
    Arithmetic throughput metrics measuring integer operations (\gls{mops}) and floating-point operations (\gls{mflops}) per second:
    \begin{align*}
        \text{\gls{mops}} &= \frac{O}{T \cdot 10^6} & \text{\gls{mflops}} &= \frac{F}{T \cdot 10^6}
    \end{align*}
    These metrics remain flawed because they fail to capture underlying hardware acceleration differences.
\end{definition}

\subsection{Execution Time}
\begin{definition}{Execution Time}{exec-time}
    The total wall-clock time required to complete a task on a computing device. Performance is strictly inversely proportional to execution time, making it the only metric that satisfies all ideal evaluation properties.
\end{definition}

\noindent \textbf{Execution Time vs. \gls{cpu} Time:} \gls{cpu} time measures pure processor engagement and is divided into \textbf{User time} (executing application instructions) and \textbf{System time} (executing \gls{os} kernel instructions on behalf of the task).

\begin{definition}{CPU Performance Equation}{cpu-perf-eq}
    Total \gls{cpu} time is evaluated using the fundamental performance equation:
    \[
        \text{\gls{cpu} Time} = N_{\text{CLK}} \cdot T_{\text{CLK}} = \text{CPI} \cdot \text{IC} \cdot T_{\text{CLK}}
    \]
    where \(\text{IC}\) is Instruction Count, \(\text{CPI}\) is \gls{cpi}, and \(T_{\text{CLK}}\) is the clock period.
\end{definition}

\begin{definition}{Average CPI}{avg-cpi}
    Computed across distinct instruction classes:
    \[
        \text{CPI} = \frac{\sum_{i=1}^n \text{IC}_i \cdot \text{CPI}_i}{\text{IC}}
    \]
\end{definition}

\subsection{Benchmarks and Normalization}
Architectural evaluations rely on standardized benchmark suites tailored to specific domains: Desktop Computing (\gls{spec}), Servers (\gls{tpc}), and Embedded Computing (\gls{eembc}).

\begin{definition}{Geometric Mean of Normalized Execution Times}{geom-mean}
    To aggregate benchmark performance without bias toward long or short tasks, suites like \gls{spec} compute the geometric mean of normalized execution times against a reference machine:
    \[
        G = \sqrt[N]{\prod_{j=1}^N \frac{T_{e_j}}{T_{R_j}}}
    \]
\end{definition}

\section{Quantitative Principles and System Optimization}
\begin{concept}{Architecture Optimization Goals}{arch-opt}
    System design is a multi-objective optimization problem balancing execution time (\(T\)), cost (\(C\)), and power (\(P\)). Desktop systems prioritize minimizing \(T \cdot C\) under a power budget (\(P \le P_{\text{max}}\)), whereas embedded systems prioritize minimizing \(C \cdot P\) under a hard deadline (\(T \le T_{\text{max}}\)).
\end{concept}

\begin{definition}{Pareto Point}{pareto-points}
    A design configuration is Pareto-optimal if no other configuration in the design space is better across all performance, cost, and power criteria.
\end{definition}

\subsection{Improving an Existing System}
\begin{definition}{Speedup}{speedup}
    The relative performance gain achieved by an architectural enhancement \(E\):
    \[
        \text{Speedup}(E) = \frac{\text{Execution Time without } E}{\text{Execution Time with } E}
    \]
\end{definition}

\begin{theorem}{Amdahl's Law}{amdahl-law}
    The performance improvement gained from using a faster execution mode is limited by the fraction of time that the faster mode can be used (\(F\)):
    \[
        S_{\text{overall}} = \frac{1}{(1 - F) + \frac{F}{S}}
    \]
    \begin{description}
        \item[Corollary 1:] Make the common case as fast as possible.
        \item[Corollary 2 (Law of Diminishing Returns):] Additional enhancements targeting the same fraction of computation yield diminishing overall returns.
    \end{description}
\end{theorem}
